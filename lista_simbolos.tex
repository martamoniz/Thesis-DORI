
\chapter*{List of Symbols}
\begin{description}[style=unboxed]
  \item[$D$] \textendash\,Absorbed dose 
  \item[$E$] \textendash\,Effective dose
  \item[$E_b$] \textendash\,Binding energy of the electron
  \item[$E_c$] \textendash\,Energy of the Compton-scattered photon
  \item[$E_o$] \textendash\,Energy of the incident photon
  \item[$E_{pe}$] \textendash\,Energy of the photoelectron
  \item[$\varepsilon$] \textendash\,Deposited energy
  \item[$\varepsilon_{tr}$] \textendash\,Transferred energy
  \item[$H$] \textendash\,Equivalent dose
  \item[$H_p(d)$] \textendash\,Dose equivalent in soft tissue at a depth $d$
  \item[$K_{col}$] \textendash\,Collision kerma
  \item[$R$] \textendash\,Radiant energy
  \item[$\theta$] \textendash\,Angle of Compton dispersion
  \item[$\bar{W}_{air}$] \textendash\,Average energy required to produce an ion pair in air
  \item[$w_R$] \textendash\,Radiation weighting factor
  \item[$w_T$] \textendash\,Tissue weighting factor
  \item[$X$] \textendash\,Exposure
  \item[$Z$] \textendash\,Atomic number
\end{description}
